125static controller_error_e receive_terminal_command(stream_handler_t* stream_handler){
126
127 Mutex_Lock(&stream_handler->mutex); /* Lock stream */
128
129 if (fgets(stream_handler->buffer, stream_handler->size, stdin) != NULL) { /* Get terminal commands */
130
131 stream_handler->buffer[strcspn(stream_handler->buffer, "\n")] = '\0'; /* Remove \n of stream and set as 0 */
132 }
133
134 Mutex_Unlock(&stream_handler->mutex); /* Unlock stream */
135
136 return CONTROLLER_ERROR_OK; /* Return OK */
137}
