138cli_handler_error_e Cli_Handler_Check(cli_handler_t* cli_handler){
139
140 CHECK_CLI_HANDLER_PTR(cli_handler); /* Verify null poiter */
141 CHECK_CLI_HANDLER_INIT(cli_handler); /* Verify no intialized */
142
143 Mutex_Lock(&cli_handler->config.stream_rx->mutex); /* Lock stream */
144
145 do { /* Do this function in loop case multithread */
146
147 if(strlen(cli_handler->config.stream_rx->buffer) > 0){
148
149 CHECK_CLI_HANDLER_FUNC_RET(check_receive_cmd(cli_handler)); /* Verify if receive a command */
150 }
151 }
152 while (cli_handler->config.multithread == true);
153
154 Mutex_Unlock(&cli_handler->config.stream_rx->mutex); /* Unlock stream */
155
156 return CLI_HANDLER_ERROR_OK; /* Return OK */
157}

183static cli_handler_error_e check_receive_cmd(cli_handler_t* cli_handler){
184
185 bool receive_cmd = false;
186 char* saveptr;
187 char* stream_token;
188
189 stream_token = Misc_Strtok_R(cli_handler->config.stream_rx->buffer, " ", &saveptr); /* Get first command */
190
191 for (size_t i = 0; i < sizeof(Commands)/sizeof(cli_handler_commands_t*); i++){ /* Search the command */
192
193 if(strcmp(stream_token, Commands[i]->name) == 0){ /* Case is a correct command */
194
195 Commands[i]->func_ptr(cli_handler, saveptr); /* Process command recived*/
196 receive_cmd = true; /* Set flag receive to indicate cmd OK*/
197
198 break;
199 }
200 }
201
202 if(receive_cmd == false){ /* Case this command dont have in list */
203
204 Cli_Handler_Out_Msg(cli_handler, "Verify sintax command, send \"Help\" for more information.\n");
205 }
206
207 return CLI_HANDLER_ERROR_OK; /* Return OK */
208}
