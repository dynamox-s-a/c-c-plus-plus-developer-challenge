157cli_handler_error_e Cli_Handler_Check(cli_handler_t* cli_handler){
158
159 CHECK_CLI_HANDLER_PTR(cli_handler); /* Verify null poiter */
160 CHECK_CLI_HANDLER_INIT(cli_handler); /* Verify no intialized */
161
162 Mutex_Lock(&cli_handler->config.stream_rx->mutex); /* Lock stream */
163
164 do { /* Do this function in loop case multithread */
165
166 if(strlen(cli_handler->config.stream_rx->buffer) > 0){
167
168 CHECK_CLI_HANDLER_FUNC_RET(check_receive_cmd(cli_handler)); /* Verify if receive a command */
169 }
170 }
171 while (cli_handler->config.multithread == true);
172
173 Mutex_Unlock(&cli_handler->config.stream_rx->mutex);
174
175 return CLI_HANDLER_ERROR_OK;
176}

210static cli_handler_error_e check_receive_cmd(cli_handler_t* cli_handler) {
211
212 bool receive_cmd = false;
213 char* saveptr;
214 char* stream_token;
215
216 stream_token = Misc_Strtok_R(cli_handler->config.stream_rx->buffer, " ", &saveptr); /* Get first command */
217
218 for (size_t i = 0; i < sizeof(Commands)/sizeof(cli_handler_commands_t*); i++){ /* Search the command */
219
220 if(strcmp(stream_token, Commands[i]->name) == 0){ /* Case is a correct command */
221
222 Commands[i]->func_ptr(cli_handler, saveptr); /* Process command recived*/
223 receive_cmd = true; /* Set flag receive to indicate cmd OK*/
224
225 break;
226 }
227 }
228
229 if (!receive_cmd) {
230 Cli_Handler_Out_Msg(cli_handler, "Verify sintax command, send \"Help\" for more information.\n");
231 }
232
233 return CLI_HANDLER_ERROR_OK; /* Return OK */
234}

243static cli_handler_error_e command_decimal(cli_handler_t* cli_handler, char* saveptr){
244
245 uint32_t decimal;
246 char* token;
247
248 token = Misc_Strtok_R(NULL, "[]", &saveptr); /* Remove [] after command */
249
250 CHECK_CLI_HANDLER_PTR(token); /* Verify if have data to parse */
251
252 Misc_Parser_Uint(&decimal, token); /* Convert value to unsigned int */
253
254 cli_handler->config.decimal = decimal; /* Set decimal configuration */
255
256 Cli_Handler_Out_Msg(cli_handler, "New decimal resolution is %d\n", decimal);
257
258 return CLI_HANDLER_ERROR_OK; /* Return OK */
259}
