83
84 char* token;

96 while (token != NULL) {

98 save_string(logs_file_handler, token); /* Save log entry to file */
99 print_terminal(logs_file_handler, token); /* Print log entry to terminal */

102 }
103 logs_file_handler->config.stream->buffer[0] = '\0';
104 }
